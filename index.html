const API_URL = "https://script.google.com/macros/s/AKfycbwS6hBrm4JOX6eQkTAMhlw6tEYIoJE5eIS6bsLOOexNjJ0gIGOXSWixoD-0WOlTmm8R/exec"; let currentActiveRate = 0; let todayRate = 0; let todayDateStr = ""; let historicoData = []; let currentMode = "today"; // "today" o "history" const usdInput = document.getElementById('usdInput'); const vesInput = document.getElementById('vesInput'); const usdLivePreview = document.getElementById('usdLivePreview'); const vesLivePreview = document.getElementById('vesLivePreview'); const rateDisplay = document.getElementById('rateDisplay'); const rateLabel = document.getElementById('rateLabel'); const dateAppliedDisplay = document.getElementById('dateAppliedDisplay'); const statusBadge = document.getElementById('statusBadge'); const btnRefresh = document.getElementById('btnRefresh'); const tabHoyBtn = document.getElementById('tabHoyBtn'); const tabHistorialBtn = document.getElementById('tabHistorialBtn'); const historyDateGroup = document.getElementById('historyDateGroup'); const historyDateInput = document.getElementById('historyDateInput'); document.addEventListener('DOMContentLoaded', () => { // 1. Cargar caché local para mostrar algo de inmediato const cachedRate = localStorage.getItem('bcv_rate_d1'); if (cachedRate) { currentActiveRate = parseFloat(cachedRate); todayRate = currentActiveRate; rateDisplay.textContent = `${currentActiveRate.toFixed(2)} Bs.`; statusBadge.textContent = "Caché Local"; } // Intentar cargar histórico guardado previamente en localStorage const cachedHistorico = localStorage.getItem('bcv_historico'); if (cachedHistorico) { try { historicoData = JSON.parse(cachedHistorico); } catch (e) { console.error("Error leyendo caché de histórico", e); } } // 2. Carga rápida de la tasa actual (D1) obtenerTasaD1(); // 3. Carga del histórico de forma silenciosa y asíncrona en segundo plano cargarHistoricoAsincrono(); }); // 1. Petición ultrarrápida solo para la tasa de D1 async function obtenerTasaD1() { statusBadge.textContent = "Conectando..."; try { const response = await fetch(`${API_URL}?action=tasa`); const data = await response.json(); if (data && data.tasa) { let rawValue = data.tasa.toString().replace(',', '.'); todayRate = parseFloat(rawValue); if (!isNaN(todayRate)) { todayDateStr = data.fecha ? data.fecha.substring(0, 10) : 'Hoy'; // Si estamos en la pestaña de hoy, actualizamos la vista de inmediato if (currentMode === "today") { mostrarTasaHoy(); } statusBadge.textContent = "Conectado"; localStorage.setItem('bcv_rate_d1', todayRate); } else { rateDisplay.textContent = "Error"; dateAppliedDisplay.textContent = "El valor en D1 no es un número válido"; } } } catch (error) { console.error("Error obteniendo D1:", error); statusBadge.textContent = "Sin conexión"; if (todayRate > 0 && currentMode === "today") { dateAppliedDisplay.textContent = "Usando última tasa guardada"; } } } // 2. Petición asíncrona que trae todo el histórico en segundo plano async function cargarHistoricoAsincrono() { try { const response = await fetch(API_URL); // Sin ?action trae todo el historial const data = await response.json(); if (data && data.datos) { historicoData = data.datos; localStorage.setItem('bcv_historico', JSON.stringify(historicoData)); console.log(`Histórico sincronizado: ${historicoData.length} registros.`); // Si el usuario ya se había cambiado a la pestaña de histórico, refrescar la búsqueda if (currentMode === "history") { buscarTasaPorFecha(); } } } catch (error) { console.error("Error cargando el histórico en segundo plano:", error); } } function mostrarTasaHoy() { currentActiveRate = todayRate; rateLabel.textContent = "Tasa de Hoy"; if (currentActiveRate > 0) { rateDisplay.textContent = `${currentActiveRate.toFixed(2)} Bs.`; dateAppliedDisplay.textContent = `Actualizado: ${todayDateStr || 'Hoy'}`; } else { rateDisplay.textContent = "--.--"; dateAppliedDisplay.textContent = "Sin datos de la tasa de hoy"; } recalcularLive('usd'); } function buscarTasaPorFecha() { const fechaSeleccionada = historyDateInput.value; if (!fechaSeleccionada) { rateDisplay.textContent = "--.--"; dateAppliedDisplay.textContent = "Selecciona una fecha válida"; currentActiveRate = 0; recalcularLive('usd'); return; } // Buscar la fecha seleccionada en el arreglo del histórico const registroEncontrado = historicoData.find(item => { if (!item.fecha) return false; return item.fecha.substring(0, 10) === fechaSeleccionada; }); if (registroEncontrado) { currentActiveRate = parseFloat(registroEncontrado.tasa); rateLabel.textContent = `Tasa del ${fechaSeleccionada}`; rateDisplay.textContent = `${currentActiveRate.toFixed(2)} Bs.`; dateAppliedDisplay.textContent = `Registro encontrado en histórico`; } else { currentActiveRate = 0; rateLabel.textContent = `Sin registro`; rateDisplay.textContent = "--.--"; dateAppliedDisplay.textContent = `No hay tasa guardada para el ${fechaSeleccionada}`; } recalcularLive('usd'); } // Navegación de Pestañas tabHoyBtn.addEventListener('click', () => { currentMode = "today"; tabHoyBtn.classList.add('active'); tabHistorialBtn.classList.remove('active'); historyDateGroup.style.display = "none"; mostrarTasaHoy(); }); tabHistorialBtn.addEventListener('click', () => { currentMode = "history"; tabHistorialBtn.classList.add('active'); tabHoyBtn.classList.remove('active'); historyDateGroup.style.display = "flex"; // Si el input no tiene fecha seleccionada, poner la de hoy por defecto if (!historyDateInput.value) { const now = new Date(); const year = now.getFullYear(); const month = String(now.getMonth() + 1).padStart(2, '0'); const day = String(now.getDate()).padStart(2, '0'); historyDateInput.value = `${year}-${month}-${day}`; } buscarTasaPorFecha(); }); historyDateInput.addEventListener('change', buscarTasaPorFecha); function recalcularLive(origen) { if (currentActiveRate <= 0) { usdLivePreview.classList.remove('visible'); vesLivePreview.classList.remove('visible'); return; } if (origen === 'usd') { const valUSD = parseFloat(usdInput.value); if (!isNaN(valUSD) && valUSD > 0) { const resultadoBs = valUSD * currentActiveRate; vesInput.value = resultadoBs.toFixed(2); usdLivePreview.textContent = `= ${resultadoBs.toLocaleString('es-VE', { minimumFractionDigits: 2, maximumFractionDigits: 2 })} Bs.`; usdLivePreview.classList.add('visible'); vesLivePreview.classList.remove('visible'); } else { vesInput.value = ''; usdLivePreview.classList.remove('visible'); } } else if (origen === 'ves') { const valVES = parseFloat(vesInput.value); if (!isNaN(valVES) && valVES > 0) { const resultadoUSD = valVES / currentActiveRate; usdInput.value = resultadoUSD.toFixed(2); vesLivePreview.textContent = `= $${resultadoUSD.toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 })}`; vesLivePreview.classList.add('visible'); usdLivePreview.classList.remove('visible'); } else { usdInput.value = ''; vesLivePreview.classList.remove('visible'); } } } usdInput.addEventListener('input', () => recalcularLive('usd')); vesInput.addEventListener('input', () => recalcularLive('ves')); btnRefresh.addEventListener('click', obtenerTasaD1); // Registro del Service Worker if ('serviceWorker' in navigator) { window.addEventListener('load', () => { navigator.serviceWorker.register('./sw.js') .then(reg => console.log('SW registrado:', reg.scope)) .catch(err => console.error('Error en SW:', err)); }); }
